    <script>
        // FIXED: Explicitly accept button element parameter to avoid cross-browser global event errors
        function filterProjects(status, btnElement) {
            
            // Update UI filter button states cleanly
            const buttons = document.querySelectorAll('.filter-btn');
            buttons.forEach(btn => {
                btn.classList.remove('btn-teal', 'active');
                btn.classList.add('btn-outline-secondary');
            });
            
            // Toggle active state on clicked element
            btnElement.classList.remove('btn-outline-secondary');
            btnElement.classList.add('btn-teal', 'active');

            // Handle card node visibility transitions dynamically
            const cards = document.querySelectorAll('.project-card-item');
            cards.forEach(card => {
                const targetStatus = card.getAttribute('data-status');
                
                if (status === 'All' || targetStatus === status) {
                    // FIXED: Using Bootstrap display utilities instead of raw block layout to preserve grid styling
                    card.classList.remove('d-none');
                    card.classList.add('animate__animated', 'animate__fadeIn');
                } else {
                    card.classList.add('d-none');
                    card.classList.remove('animate__animated', 'animate__fadeIn');
                }
            });
        }
    </script>
